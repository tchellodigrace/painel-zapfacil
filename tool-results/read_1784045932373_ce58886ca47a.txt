    74→function formatarData(data: string) {
    75→  if (!data) return "";
    76→  const partes = data.split("-");
    77→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    78→  return data;
    79→}

    81→function diasRestantes(vencimento: string): number {
    82→  const hoje = new Date();
    83→  hoje.setHours(0, 0, 0, 0);
    84→  const venc = new Date(vencimento + "T00:00:00");
    85→  const diff = venc.getTime() - hoje.getTime();
    86→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    87→}

    92→function DialogTrocarSenha({
    93→  open,
    94→  onOpenChange,
    95→}: {
    96→  open: boolean;
    97→  onOpenChange: (v: boolean) => void;
    98→}) {
    99→  const { alterarSenha } = useAdminStore();
   100→  const [senhaAtual, setSenhaAtual] = useState("");
   101→  const [novaSenha, setNovaSenha] = useState("");
   102→  const [confirmarNova, setConfirmarNova] = useState("");
   103→  const [mostrarAtual, setMostrarAtual] = useState(false);
   104→  const [mostrarNova, setMostrarNova] = useState(false);
   105→
   106→  const handleSalvar = () => {
   107→    if (!senhaAtual || !novaSenha) {
   108→      toast.error("Preencha todos os campos.");
   109→      return;
   110→    }
   111→    if (novaSenha.length < 4) {
   112→      toast.error("A nova senha deve ter pelo menos 4 caracteres.");
   113→      return;
   114→    }
   115→    if (novaSenha !== confirmarNova) {
   116→      toast.error("As senhas nao conferem.");
   117→      return;
   118→    }
   119→    const ok = alterarSenha(senhaAtual, novaSenha);
   120→    if (ok) {
   121→      toast.success("Senha alterada com sucesso!");
   122→      setSenhaAtual("");
   123→      setNovaSenha("");
   124→      setConfirmarNova("");
   125→      onOpenChange(false);
   126→    } else {
   127→      toast.error("Senha atual incorreta.");
   128→    }
   129→  };
   130→
   131→  return (
   132→    <Dialog open={open} onOpenChange={onOpenChange}>
   133→      <DialogContent className="max-w-sm">
   134→        <DialogHeader>
   135→          <DialogTitle className="flex items-center gap-2 text-base">
   136→            <KeyRound className="h-5 w-5 text-gray-600" />
   137→            Alterar Senha
   138→          </DialogTitle>
   139→        </DialogHeader>
   140→        <div className="space-y-4">
   141→          <div className="space-y-1.5">
   142→            <Label className="text-xs font-medium">Senha atual</Label>
   143→            <div className="relative">
   144→              <Input
   145→                type={mostrarAtual ? "text" : "password"}
   146→                value={senhaAtual}
   147→                onChange={(e) => setSenhaAtual(e.target.value)}
   148→                placeholder="Digite a senha atual"
   149→                className="pr-10 h-10 text-sm"
   150→                onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   151→              />
   152→              <button
   153→                type="button"
   154→                onClick={() => setMostrarAtual(!mostrarAtual)}
   155→                className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600"
   156→              >
   157→                {mostrarAtual ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   158→              </button>
   159→            </div>
   160→          </div>
   161→          <div className="space-y-1.5">
   162→            <Label className="text-xs font-medium">Nova senha</Label>
   163→            <Input
   164→              type="password"
   165→              value={novaSenha}
   166→              onChange={(e) => setNovaSenha(e.target.value)}
   167→              placeholder="Minimo 4 caracteres"
   168→              className="h-10 text-sm"
   169→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   170→            />
   171→          </div>
   172→          <div className="space-y-1.5">
   173→            <Label className="text-xs font-medium">Confirmar nova senha</Label>
   174→            <Input
   175→              type="password"
   176→              value={confirmarNova}
   177→              onChange={(e) => setConfirmarNova(e.target.value)}
   178→              placeholder="Repita a nova senha"
   179→              className="h-10 text-sm"
   180→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   181→            />
   182→          </div>
   183→          <div className="flex gap-2 justify-end pt-1">
   184→            <Button variant="ghost" size="sm" className="text-xs" onClick={() => onOpenChange(false)}>
   185→              Cancelar
   186→            </Button>
   187→            <Button size="sm" className="text-xs bg-emerald-600 hover:bg-emerald-700" onClick={handleSalvar}>
   188→              Salvar
   189→            </Button>
   190→          </div>
   191→        </div>
   192→      </DialogContent>
   193→    </Dialog>
   194→  );
   195→}
