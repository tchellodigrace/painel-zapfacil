    68→function formatarData(data: string) {
    69→  if (!data) return "";
    70→  const partes = data.split("-");
    71→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    72→  return data;
    73→}

    75→function diasRestantes(vencimento: string): number {
    76→  const hoje = new Date();
    77→  hoje.setHours(0, 0, 0, 0);
    78→  const venc = new Date(vencimento + "T00:00:00");
    79→  const diff = venc.getTime() - hoje.getTime();
    80→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    81→}

    86→function DialogTrocarSenha({
    87→  open,
    88→  onOpenChange,
    89→}: {
    90→  open: boolean;
    91→  onOpenChange: (v: boolean) => void;
    92→}) {
    93→  const { alterarSenha } = useAdminStore();
    94→  const [senhaAtual, setSenhaAtual] = useState("");
    95→  const [novaSenha, setNovaSenha] = useState("");
    96→  const [confirmarNova, setConfirmarNova] = useState("");
    97→  const [mostrarAtual, setMostrarAtual] = useState(false);
    98→  const [mostrarNova, setMostrarNova] = useState(false);
    99→
   100→  const handleSalvar = () => {
   101→    if (!senhaAtual || !novaSenha) {
   102→      toast.error("Preencha todos os campos.");
   103→      return;
   104→    }
   105→    if (novaSenha.length < 4) {
   106→      toast.error("A nova senha deve ter pelo menos 4 caracteres.");
   107→      return;
   108→    }
   109→    if (novaSenha !== confirmarNova) {
   110→      toast.error("As senhas nao conferem.");
   111→      return;
   112→    }
   113→    const ok = alterarSenha(senhaAtual, novaSenha);
   114→    if (ok) {
   115→      toast.success("Senha alterada com sucesso!");
   116→      setSenhaAtual("");
   117→      setNovaSenha("");
   118→      setConfirmarNova("");
   119→      onOpenChange(false);
   120→    } else {
   121→      toast.error("Senha atual incorreta.");
   122→    }
   123→  };
   124→
   125→  return (
   126→    <Dialog open={open} onOpenChange={onOpenChange}>
   127→      <DialogContent className="max-w-sm">
   128→        <DialogHeader>
   129→          <DialogTitle className="flex items-center gap-2 text-base">
   130→            <KeyRound className="h-5 w-5 text-gray-600" />
   131→            Alterar Senha
   132→          </DialogTitle>
   133→        </DialogHeader>
   134→        <div className="space-y-4">
   135→          <div className="space-y-1.5">
   136→            <Label className="text-xs font-medium">Senha atual</Label>
   137→            <div className="relative">
   138→              <Input
   139→                type={mostrarAtual ? "text" : "password"}
   140→                value={senhaAtual}
   141→                onChange={(e) => setSenhaAtual(e.target.value)}
   142→                placeholder="Digite a senha atual"
   143→                className="pr-10 h-10 text-sm"
   144→                onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   145→              />
   146→              <button
   147→                type="button"
   148→                onClick={() => setMostrarAtual(!mostrarAtual)}
   149→                className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600"
   150→              >
   151→                {mostrarAtual ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   152→              </button>
   153→            </div>
   154→          </div>
   155→          <div className="space-y-1.5">
   156→            <Label className="text-xs font-medium">Nova senha</Label>
   157→            <Input
   158→              type="password"
   159→              value={novaSenha}
   160→              onChange={(e) => setNovaSenha(e.target.value)}
   161→              placeholder="Minimo 4 caracteres"
   162→              className="h-10 text-sm"
   163→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   164→            />
   165→          </div>
   166→          <div className="space-y-1.5">
   167→            <Label className="text-xs font-medium">Confirmar nova senha</Label>
   168→            <Input
   169→              type="password"
   170→              value={confirmarNova}
   171→              onChange={(e) => setConfirmarNova(e.target.value)}
   172→              placeholder="Repita a nova senha"
   173→              className="h-10 text-sm"
   174→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   175→            />
   176→          </div>
