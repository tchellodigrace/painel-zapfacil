    63→    63→function formatarData(data: string) {
    64→    64→  if (!data) return "";
    65→    65→  const partes = data.split("-");
    66→    66→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    67→    67→  return data;

    70→    70→function diasRestantes(vencimento: string): number {
    71→    71→  const hoje = new Date();
    72→    72→  hoje.setHours(0, 0, 0, 0);
    73→    73→  const venc = new Date(vencimento + "T00:00:00");
    74→    74→  const diff = venc.getTime() - hoje.getTime();
    75→    75→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    76→    76→}
