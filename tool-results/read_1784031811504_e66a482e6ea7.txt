    38→    38→function carregarCredenciais(): Credenciais | null {
    39→    39→  if (typeof window === "undefined") return null;
    40→    40→  try {
    41→    41→    const item = localStorage.getItem(AUTH_KEY);
    42→    42→    return item ? (JSON.parse(item) as Credenciais) : null;
    43→    43→  } catch {
    44→    44→    return null;
    45→    45→  }
    46→    46→}
    47→    47→
    48→    48→function salvarCredenciais(cred: Credenciais): void {
    49→    49→  if (typeof window === "undefined") return;
    50→    50→  localStorage.setItem(AUTH_KEY, JSON.stringify(cred));
    51→    51→}
    52→    52→
    53→    53→function criarSessao(): void {
    54→    54→  if (typeof window === "undefined") return;
    55→    55→  sessionStorage.setItem(SESSION_KEY, "autenticado");
    56→    56→}
    57→    57→
    58→    58→function verificarSessao(): boolean {
    59→    59→  if (typeof window === "undefined") return false;
    60→    60→  return sessionStorage.getItem(SESSION_KEY) === "autenticado";
    61→    61→}
    62→    62→
    63→    63→function destruirSessao(): void {
    64→    64→  if (typeof window === "undefined") return;
    65→    65→  sessionStorage.removeItem(SESSION_KEY);
    66→    66→}
