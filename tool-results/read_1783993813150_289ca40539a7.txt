    61→    61→function formatarData(data: string) {
    62→    62→  if (!data) return "";
    63→    63→  const partes = data.split("-");
    64→    64→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    65→    65→  return data;
    66→    66→}

    68→    68→function diasRestantes(vencimento: string): number {
    69→    69→  const hoje = new Date();
    70→    70→  hoje.setHours(0, 0, 0, 0);
    71→    71→  const venc = new Date(vencimento + "T00:00:00");
    72→    72→  const diff = venc.getTime() - hoje.getTime();
    73→    73→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    74→    74→}
