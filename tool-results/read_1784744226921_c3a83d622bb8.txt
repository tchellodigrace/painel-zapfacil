    74→function formatarData(data: string) {
    75→  if (!data) return "";
    76→  const partes = data.split("-");
    77→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;

    81→function diasRestantes(vencimento: string): number {
    82→  const hoje = new Date();
    83→  hoje.setHours(0, 0, 0, 0);
    84→  const venc = new Date(vencimento + "T00:00:00");
    85→  const diff = venc.getTime() - hoje.getTime();
    86→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    87→}

    92→function DialogTrocarSenha({
    93→  open,
    94→  onOpenChange,
    95→}: {
    96→  open: boolean;
    97→  onOpenChange: (v: boolean) => void;
    98→}) {
    99→  const { alterarSenha } = useAdminStore();
