    66→function formatarData(data: string) {
    67→  if (!data) return "";
    68→  const partes = data.split("-");
    69→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    70→  return data;
    71→}

    73→function diasRestantes(vencimento: string): number {
    74→  const hoje = new Date();
    75→  hoje.setHours(0, 0, 0, 0);
    76→  const venc = new Date(vencimento + "T00:00:00");
    77→  const diff = venc.getTime() - hoje.getTime();
    78→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    79→}

   464→function PainelAdminConteudo() {
   465→  const { sistemas, adicionarSistema, editarSistema, removerSistema, alterarStatus } =
   466→    useAdminStore();
   467→  const [busca, setBusca] = useState("");

   469→  const [filtroPlano, setFiltroPlano] = useState<string>("TODOS");
   470→  const [dialogForm, setDialogForm] = useState<SistemaCliente | null>(null);
   471→  const [dialogNovo, setDialogNovo] = useState(false);
   472→  const [dialogDetalhe, setDialogDetalhe] = useState<SistemaCliente | null>(null);
   473→  const [confirmaRemover, setConfirmaRemover] = useState<string | null>(null);
   474→
   475→  // Estatísticas
   476→  const stats = useMemo(() => {
   477→    const ativos = sistemas.filter((s) => s.status === "ATIVO").length;
   478→    const trials = sistemas.filter((s) => s.status === "TRIAL").length;
   479→    const expirados = sistemas.filter((s) => s.status === "EXPIRADO").length;
   480→    const cancelados = sistemas.filter((s) => s.status === "CANCELADO").length;
   481→    const receitaMensal = sistemas
   482→      .filter((s) => s.status === "ATIVO" || s.status === "TRIAL")
   483→      .reduce((s, v) => s + v.valorMensal, 0);
   484→    const vencendo = sistemas.filter(
   485→      (s) => s.status === "ATIVO" && diasRestantes(s.dataVencimento) <= 7 && diasRestantes(s.dataVencimento) > 0
   486→    ).length;
   487→    return { ativos, trials, expirados, cancelados, receitaMensal, vencendo, total: sistemas.length };
   488→  }, [sistemas]);
   489→
   490→  // Filtragem
   491→  const sistemasFiltrados = useMemo(() => {
   492→    let lista = sistemas;
   493→    if (filtroStatus !== "TODOS") {
   494→      lista = lista.filter((s) => s.status === filtroStatus);
   495→    }
   496→    if (filtroPlano !== "TODOS") {
   497→      lista = lista.filter((s) => s.plano === filtroPlano);
   498→    }
   499→    if (busca.trim()) {
   500→      const termo = busca.toLowerCase();
   501→      lista = lista.filter(
   502→        (s) =>
   503→          s.empresa.toLowerCase().includes(termo) ||
   504→          s.responsavel.toLowerCase().includes(termo) ||
   505→          s.cidade.toLowerCase().includes(termo) ||
   506→          s.telefone.includes(termo) ||
   507→          s.email.toLowerCase().includes(termo)
   508→      );
   509→    }
   510→    return lista;
   511→  }, [sistemas, filtroStatus, filtroPlano, busca]);
   512→
   513→  const handleSalvarNovo = useCallback(
   514→    (dados: Omit<SistemaCliente, "id" | "criadoEm">) => {
   515→      adicionarSistema(dados);
   516→      setDialogNovo(false);
   517→      toast.success("Sistema cadastrado!");
   518→    },
   519→    [adicionarSistema]
   520→  );
   521→
   522→  const handleSalvarEdicao = useCallback(
   523→    (dados: Omit<SistemaCliente, "id" | "criadoEm">) => {
   524→      if (!dialogForm) return;
   525→      editarSistema(dialogForm.id, dados);
   526→      setDialogForm(null);
   527→      toast.success("Sistema atualizado!");
   528→    },
   529→    [dialogForm, editarSistema]
   530→  );
   531→
   532→  const handleRemover = useCallback(
   533→    (id: string) => {
   534→      removerSistema(id);
   535→      setConfirmaRemover(null);
   536→      toast.success("Sistema removido.");
   537→    },
   538→    [removerSistema]
   539→  );
   540→
   541→  const handleWhatsApp = (telefone: string) => {
   542→    const telLimpo = telefone.replace(/\D/g, "");
   543→    const numero = telLimpo.startsWith("55") ? telLimpo : `55${telLimpo}`;
   544→    window.open(`https://wa.me/${numero}`, "_blank");
   545→  };
   546→
   547→  const handleLogout = () => {
   548→    sessionStorage.removeItem("zapfacil_admin_session");
   549→    window.location.reload();
   550→  };
   551→
   552→  const getStatusInfo = (status: StatusSistema) =>
   553→    STATUS_SISTEMA.find((s) => s.valor === status) || STATUS_SISTEMA[3];
   554→  const getPlanoInfo = (plano: PlanoSistema) =>
   555→    PLANOS.find((p) => p.valor === plano) || PLANOS[0];
