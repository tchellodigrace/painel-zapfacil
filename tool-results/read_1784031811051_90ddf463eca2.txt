    75→    75→function diasRestantes(vencimento: string): number {
    76→    76→  const hoje = new Date();
    77→    77→  hoje.setHours(0, 0, 0, 0);
    78→    78→  const venc = new Date(vencimento + "T00:00:00");
    79→    79→  const diff = venc.getTime() - hoje.getTime();
    80→    80→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    81→    81→}

   638→   638→  function buscarCredenciaisCliente(email: string) {
   639→   639→    // Procura o sistema que tem registro com esse email
   640→   640→    const sistema = sistemas.find(
   641→   641→      (s) =>
   642→   642→        s.email.toLowerCase() === email.toLowerCase() ||
   643→   643→        s.dadosRegistro?.email.toLowerCase() === email.toLowerCase()
   644→   644→    );
   645→   645→    if (sistema?.dadosRegistro) {
   646→   646→      // Se o cliente se registrou, procura as credenciais no localStorage do ERP
   647→   647→      // Como as credenciais ficam no dispositivo do cliente, o admin precisa informar manualmente
   648→   648→      // Mas podemos retornar os dados do registro
   649→   649→      return {
   650→   650→        nome: sistema.dadosRegistro.usuario || sistema.responsavel,
   651→   651→        empresa: sistema.dadosRegistro.nomeEmpresa || sistema.empresa,
   652→   652→        telefone: sistema.dadosRegistro.telefone || sistema.telefone || "",
   653→   653→        email: sistema.dadosRegistro.email || sistema.email,
   654→   654→      };
   655→   655→    }
   656→   656→    // Fallback: busca pelo email nos sistemas
   657→   657→    const s = sistemas.find((s) => s.email.toLowerCase() === email.toLowerCase());
   658→   658→    if (s) {
   659→   659→      return {
   660→   660→        nome: s.responsavel,
   661→   661→        empresa: s.empresa,
   662→   662→        telefone: s.telefone,
   663→   663→        email: s.email,
   664→   664→      };
   665→   665→    }
   666→   666→    return null;
