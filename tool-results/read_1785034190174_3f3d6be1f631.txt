    81→function formatarData(data: string) {
    82→  if (!data) return "";
    83→  const partes = data.split("-");
    84→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    85→  return data;
    86→}

    88→function diasRestantes(vencimento: string): number {
    89→  const hoje = new Date();
    90→  hoje.setHours(0, 0, 0, 0);
    91→  const venc = new Date(vencimento + "T00:00:00");
    92→  const diff = venc.getTime() - hoje.getTime();
    93→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    94→}

    99→function DialogTrocarSenha({
   100→  open,
   101→  onOpenChange,
   102→}: {
   103→  open: boolean;
   104→  onOpenChange: (v: boolean) => void;
   105→}) {
   106→  const { alterarSenha } = useAdminStore();
   107→  const [senhaAtual, setSenhaAtual] = useState("");
   108→  const [novaSenha, setNovaSenha] = useState("");
   109→  const [confirmarNova, setConfirmarNova] = useState("");
   110→  const [mostrarAtual, setMostrarAtual] = useState(false);
   111→  const [mostrarNova, setMostrarNova] = useState(false);
   112→
   113→  const handleSalvar = () => {
   114→    if (!senhaAtual || !novaSenha) {
   115→      toast.error("Preencha todos os campos.");
   116→      return;
   117→    }
   118→    if (novaSenha.length < 4) {
   119→      toast.error("A nova senha deve ter pelo menos 4 caracteres.");
   120→      return;
   121→    }
   122→    if (novaSenha !== confirmarNova) {
   123→      toast.error("As senhas nao conferem.");
   124→      return;
   125→    }
   126→    const ok = alterarSenha(senhaAtual, novaSenha);
   127→    if (ok) {
   128→      toast.success("Senha alterada com sucesso!");
   129→      setSenhaAtual("");
   130→      setNovaSenha("");
   131→      setConfirmarNova("");
   132→      onOpenChange(false);
   133→    } else {
   134→      toast.error("Senha atual incorreta.");
   135→    }
   136→  };
   137→
   138→  return (
   139→    <Dialog open={open} onOpenChange={onOpenChange}>
   140→      <DialogContent className="max-w-sm">
   141→        <DialogHeader>
   142→          <DialogTitle className="flex items-center gap-2 text-base">
   143→            <KeyRound className="h-5 w-5 text-gray-600" />
   144→            Alterar Senha
   145→          </DialogTitle>
   146→        </DialogHeader>
   147→        <div className="space-y-4">
   148→          <div className="space-y-1.5">
   149→            <Label className="text-xs font-medium">Senha atual</Label>
   150→            <div className="relative">
   151→              <Input
   152→                type={mostrarAtual ? "text" : "password"}
   153→                value={senhaAtual}
   154→                onChange={(e) => setSenhaAtual(e.target.value)}
   155→                placeholder="Digite a senha atual"
   156→                className="pr-10 h-10 text-sm"
   157→                onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   158→              />
   159→              <button
   160→                type="button"
   161→                onClick={() => setMostrarAtual(!mostrarAtual)}
   162→                className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600"
   163→              >
   164→                {mostrarAtual ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   165→              </button>
   166→            </div>
   167→          </div>
   168→          <div className="space-y-1.5">
   169→            <Label className="text-xs font-medium">Nova senha</Label>
   170→            <Input
   171→              type="password"
   172→              value={novaSenha}
   173→              onChange={(e) => setNovaSenha(e.target.value)}
   174→              placeholder="Minimo 4 caracteres"
   175→              className="h-10 text-sm"
   176→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   177→            />
   178→          </div>
   179→          <div className="space-y-1.5">
   180→            <Label className="text-xs font-medium">Confirmar nova senha</Label>
   181→            <Input
   182→              type="password"
   183→              value={confirmarNova}
   184→              onChange={(e) => setConfirmarNova(e.target.value)}
   185→              placeholder="Repita a nova senha"
   186→              className="h-10 text-sm"
   187→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   188→            />
   189→          </div>
   190→          <div className="flex gap-2 justify-end pt-1">
   191→            <Button variant="ghost" size="sm" className="text-xs" onClick={() => onOpenChange(false)}>
   192→              Cancelar
   193→            </Button>
   194→            <Button size="sm" className="text-xs bg-emerald-600 hover:bg-emerald-700" onClick={handleSalvar}>
   195→              Salvar
   196→            </Button>
   197→          </div>
   198→        </div>
   199→      </DialogContent>
   200→    </Dialog>
   201→  );
   202→}
