    29→    29→interface Credenciais {

    37→    37→function carregarCredenciais(): Credenciais | null {
    38→    38→  if (typeof window === "undefined") return null;
    39→    39→  try {
    40→    40→    const item = localStorage.getItem(AUTH_KEY);
    41→    41→    return item ? (JSON.parse(item) as Credenciais) : null;
    42→    42→  } catch {
    43→    43→    return null;
    44→    44→  }
    45→    45→}
    46→    46→
    47→    47→function salvarCredenciais(cred: Credenciais): void {
    48→    48→  if (typeof window === "undefined") return;
    49→    49→  localStorage.setItem(AUTH_KEY, JSON.stringify(cred));
    50→    50→}
    51→    51→
    52→    52→function criarSessao(): void {
    53→    53→  if (typeof window === "undefined") return;
    54→    54→  sessionStorage.setItem(SESSION_KEY, "autenticado");
    55→    55→}
    56→    56→
    57→    57→function verificarSessao(): boolean {
    58→    58→  if (typeof window === "undefined") return false;
    59→    59→  return sessionStorage.getItem(SESSION_KEY) === "autenticado";
    60→    60→}
    61→    61→
    62→    62→function destruirSessao(): void {
    63→    63→  if (typeof window === "undefined") return;
    64→    64→  sessionStorage.removeItem(SESSION_KEY);
    65→    65→}
