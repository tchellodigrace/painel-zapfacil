    26→const AUTH_KEY = "zapfacil_auth";
    27→const SESSION_KEY = "zapfacil_session";
    28→
    29→interface Credenciais {
    30→  email: string;
    31→  senha: string;
    32→  nomeEmpresa: string;
    33→  nomeResponsavel: string;
    34→  telefone: string;
    35→}
    36→
    37→function carregarCredenciais(): Credenciais | null {
    38→  if (typeof window === "undefined") return null;
    39→  try {
    40→    const item = localStorage.getItem(AUTH_KEY);
    41→    return item ? (JSON.parse(item) as Credenciais) : null;
    42→  } catch {
    43→    return null;
    44→  }
    45→}
    46→
    47→function salvarCredenciais(cred: Credenciais): void {
    48→  if (typeof window === "undefined") return;
    49→  localStorage.setItem(AUTH_KEY, JSON.stringify(cred));
    50→}
    51→
    52→function criarSessao(): void {
    53→  if (typeof window === "undefined") return;
    54→  sessionStorage.setItem(SESSION_KEY, "autenticado");
    55→}
    56→
    57→function verificarSessao(): boolean {
    58→  if (typeof window === "undefined") return false;
    59→  return sessionStorage.getItem(SESSION_KEY) === "autenticado";
    60→}
    61→
    62→function destruirSessao(): void {
    63→  if (typeof window === "undefined") return;
    64→  sessionStorage.removeItem(SESSION_KEY);
    65→}
