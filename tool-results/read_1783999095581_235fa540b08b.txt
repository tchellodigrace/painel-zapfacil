    59→function formatarMoeda(valor: number) {
    60→  return valor.toLocaleString("pt-BR", { style: "currency", currency: "BRL" });
    61→}
    62→
    63→function formatarData(data: string) {
    64→  if (!data) return "";
    65→  const partes = data.split("-");
    66→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    67→  return data;
    68→}

    70→function diasRestantes(vencimento: string): number {
    71→  const hoje = new Date();
    72→  hoje.setHours(0, 0, 0, 0);
    73→  const venc = new Date(vencimento + "T00:00:00");
    74→  const diff = venc.getTime() - hoje.getTime();
    75→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    76→}

    81→function FormularioCobranca({
    82→  sistema,
    83→  cobrancaEdit,
    84→  onSalvar,
    85→  onCancelar,
    86→}: {
    87→  sistema?: SistemaCliente;
    88→  cobrancaEdit?: Cobranca;
    89→  onSalvar: (dados: Omit<Cobranca, "id" | "criadoEm">) => void;
    90→  onCancelar: () => void;
    91→}) {
    92→  const { sistemas } = useAdminStore();
    93→  const sistemasAtivos = sistemas.filter(
    94→    (s) => s.status === "ATIVO" || s.status === "TRIAL"
    95→  );
    96→
    97→  const hoje = new Date().toISOString().split("T")[0];
    98→
    99→  const [sistemaId, setSistemaId] = useState(
   100→    cobrancaEdit?.sistemaId || sistema?.id || ""
   101→  );
   102→  const [tipo, setTipo] = useState<TipoCobranca>(
   103→    cobrancaEdit?.tipo || "MENSALIDADE"
   104→  );
   105→  const [descricao, setDescricao] = useState(
   106→    cobrancaEdit?.descricao || ""
   107→  );
   108→  const [valor, setValor] = useState(
   109→    cobrancaEdit?.valor?.toString() || ""
   110→  );
   111→  const [dataVencimento, setDataVencimento] = useState(
   112→    cobrancaEdit?.dataVencimento || hoje
   113→  );
   114→  const [observacoes, setObservacoes] = useState(
   115→    cobrancaEdit?.observacoes || ""
   116→  );
   117→
   118→  // Auto-preenche quando seleciona sistema
   119→  const sistemaSel = sistemasAtivos.find((s) => s.id === sistemaId);
   120→
   121→  const handleSelectSistema = (id: string) => {
   122→    setSistemaId(id);
   123→    const s = sistemasAtivos.find((x) => x.id === id);
   124→    if (s && !cobrancaEdit) {
   125→      if (s.tipoLicenca === "ALUGUEL") {
   126→        setTipo("MENSALIDADE");
   127→        setValor(s.valorMensal?.toString() || "");
   128→      } else {
   129→        setTipo("AQUISICAO");
   130→        setValor(s.valorAquisicao?.toString() || "");
   131→      }
   132→      setDescricao("");
   133→    }
   134→  };
   135→
   136→  const handleSelectTipo = (t: string) => {
   137→    setTipo(t as TipoCobranca);
   138→    if (!cobrancaEdit && sistemaSel) {
   139→      if (t === "MENSALIDADE") setValor(sistemaSel.valorMensal?.toString() || "");
   140→      if (t === "AQUISICAO") setValor(sistemaSel.valorAquisicao?.toString() || "");
   141→      if (t === "TAXA_INSTALACAO") setValor(sistemaSel.taxaInstalacao?.toString() || "");
   142→    }
   143→  };
   144→
   145→  const handleSubmit = (e: React.FormEvent) => {
   146→    e.preventDefault();
   147→    if (!sistemaId) {
   148→      toast.error("Selecione um sistema.");
   149→      return;
   150→    }
   151→    const s = sistemas.find((x) => x.id === sistemaId);
   152→    if (!descricao.trim() && tipo === "OUTROS") {
   153→      toast.error("Descreva esta cobranca.");
   154→      return;
   155→    }
   156→    if (!dataVencimento) {
   157→      toast.error("Data de vencimento e obrigatoria.");
   158→      return;
   159→    }
   160→    const valorNum = parseFloat(valor) || 0;
   161→    if (valorNum <= 0) {
   162→      toast.error("Valor deve ser maior que zero.");
   163→      return;
   164→    }
   165→
   166→    let descFinal = descricao.trim();
   167→    if (!descFinal) {
   168→      const mesRef = new Date(dataVencimento + "T00:00:00").toLocaleString("pt-BR", {
   169→        month: "long",
   170→        year: "numeric",
   171→      });
   172→      if (tipo === "MENSALIDADE") descFinal = `Mensalidade - ${mesRef}`;
   173→      else if (tipo === "AQUISICAO") descFinal = "Aquisicao do sistema (licenca definitiva)";
   174→      else if (tipo === "TAXA_INSTALACAO") descFinal = "Taxa de instalacao";
   175→      else if (tipo === "TAXA_SUPORTE") descFinal = "Taxa de suporte tecnico";
   176→    }
   177→
   178→    onSalvar({
   179→      sistemaId,
   180→      sistemaNome: s?.empresa || "",
   181→      tipo,
   182→      descricao: descFinal,
   183→      valor: valorNum,
   184→      dataVencimento,
   185→      dataPagamento: cobrancaEdit?.dataPagamento || null,
   186→      status: cobrancaEdit?.status || "PENDENTE",
   187→      formaPagamento: cobrancaEdit?.formaPagamento || null,
   188→      observacoes: observacoes.trim(),
   189→    });
   190→  };
   191→
   192→  return (
   193→    <form onSubmit={handleSubmit} className="space-y-4">
   194→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   195→        <div className="sm:col-span-2 space-y-1.5">
   196→          <Label className="text-xs font-medium">Sistema *</Label>
   197→          <Select value={sistemaId} onValueChange={handleSelectSistema} disabled={!!cobrancaEdit}>
   198→            <SelectTrigger className="text-sm h-9">
   199→              <SelectValue placeholder="Selecione o sistema" />
   200→            </SelectTrigger>
   201→            <SelectContent>
   202→              {sistemasAtivos.map((s) => (
   203→                <SelectItem key={s.id} value={s.id}>
   204→                  {s.empresa} — {s.responsavel}
   205→                </SelectItem>
   206→              ))}
   207→            </SelectContent>
   208→          </Select>
   209→        </div>
   210→
   211→        <div className="space-y-1.5">
   212→          <Label className="text-xs font-medium">Tipo de Cobranca *</Label>
   213→          <Select value={tipo} onValueChange={handleSelectTipo}>
   214→            <SelectTrigger className="text-sm h-9">
   215→              <SelectValue />
   216→            </SelectTrigger>
   217→            <SelectContent>
   218→              {TIPOS_COBRANCA.map((t) => (
   219→                <SelectItem key={t.valor} value={t.valor}>
   220→                  {t.label}
   221→                </SelectItem>
   222→              ))}
   223→            </SelectContent>
   224→          </Select>
   225→        </div>
   226→
   227→        <div className="space-y-1.5">
   228→          <Label className="text-xs font-medium">Valor (R$) *</Label>
   229→          <Input
   230→            type="number"
   231→            step="0.01"
   232→            value={valor}
   233→            onChange={(e) => setValor(e.target.value)}
   234→            placeholder="0,00"
   235→            className="text-sm h-9"
   236→          />
   237→        </div>
   238→
   239→        <div className="space-y-1.5">
   240→          <Label className="text-xs font-medium">Vencimento *</Label>
   241→          <Input
   242→            type="date"
   243→            value={dataVencimento}
   244→            onChange={(e) => setDataVencimento(e.target.value)}
   245→            className="text-sm h-9"
   246→          />
   247→        </div>
   248→
   249→        <div className="space-y-1.5">
   250→          <Label className="text-xs font-medium">Descricao</Label>
   251→          <Input
   252→            value={descricao}
   253→            onChange={(e) => setDescricao(e.target.value)}
   254→            placeholder="Auto-preenchido pelo tipo"
   255→            className="text-sm h-9"
   256→          />
   257→        </div>
   258→
   259→        <div className="sm:col-span-2 space-y-1.5">
   260→          <Label className="text-xs font-medium">Observacoes</Label>
   261→          <Textarea
   262→            value={observacoes}
   263→            onChange={(e) => setObservacoes(e.target.value)}
   264→            placeholder="Notas internas..."
   265→            className="text-sm min-h-[50px]"
   266→          />
   267→        </div>
   268→      </div>
   269→      <div className="flex gap-2 justify-end pt-2">
   270→        <Button
   271→          type="button"
   272→          variant="ghost"
   273→          size="sm"
   274→          className="text-xs"
   275→          onClick={onCancelar}
   276→        >
   277→          Cancelar
   278→        </Button>
   279→        <Button
   280→          type="submit"
   281→          size="sm"
   282→          className="text-xs bg-emerald-600 hover:bg-emerald-700"
   283→        >
   284→          {cobrancaEdit ? "Salvar Alteracoes" : "Gerar Cobranca"}
   285→        </Button>
   286→      </div>
   287→    </form>
   288→  );
   289→}

   365→export function PainelCobranças() {
   366→  const {
   367→    cobrancas,
   368→    sistemas,
   369→    adicionarCobranca,
   370→    editarCobranca,
   371→    removerCobranca,
   372→    registrarPagamento,
   373→    cancelarCobranca,
   374→    gerarCobrancaMensal,
   375→    gerarCobrancaAquisicao,
   376→  } = useAdminStore();
   377→
   378→  const [busca, setBusca] = useState("");
   379→  const [filtroStatus, setFiltroStatus] = useState<string>("TODOS");
   380→  const [filtroTipo, setFiltroTipo] = useState<string>("TODOS");
   381→  const [filtroSistema, setFiltroSistema] = useState<string>("TODOS");
   382→  const [dialogNova, setDialogNova] = useState(false);
   383→  const [dialogPagamento, setDialogPagamento] = useState<Cobranca | null>(null);
   384→  const [dialogEditCobranca, setDialogEditCobranca] = useState<Cobranca | null>(null);
   385→  const [dialogHistorico, setDialogHistorico] = useState<SistemaCliente | null>(null);
   386→  const [confirmaRemoverCob, setConfirmaRemoverCob] = useState<string | null>(null);
   387→
   388→  // Estatísticas
   389→  const stats = useMemo(() => {
   390→    const agora = new Date();
   391→    const mesAtual = agora.getMonth();
   392→    const anoAtual = agora.getFullYear();
   393→
   394→    const pagosMes = cobrancas.filter((c) => {
   395→      if (c.status !== "PAGO" || !c.dataPagamento) return false;
   396→      const dp = new Date(c.dataPagamento + "T00:00:00");
   397→      return dp.getMonth() === mesAtual && dp.getFullYear() === anoAtual;
   398→    });
   399→
   400→    const pendentes = cobrancas.filter((c) => c.status === "PENDENTE");
   401→    const atrasados = cobrancas.filter((c) => c.status === "ATRASADO");
   402→    const totalRecebidoMes = pagosMes.reduce((s, c) => s + c.valor, 0);
   403→    const totalPendente = pendentes.reduce((s, c) => s + c.valor, 0);
   404→    const totalAtrasado = atrasados.reduce((s, c) => s + c.valor, 0);
   405→    const previstoMes = cobrancas
   406→      .filter((c) => {
   407→        if (c.status === "CANCELADO") return false;
   408→        const venc = new Date(c.dataVencimento + "T00:00:00");
   409→        return venc.getMonth() === mesAtual && venc.getFullYear() === anoAtual;
   410→      })
   411→      .reduce((s, c) => s + c.valor, 0);
   412→
   413→    return {
   414→      recebidoMes: totalRecebidoMes,
   415→      pendentes: pendentes.length,
   416→      atrasados: atrasados.length,
   417→      totalPendente,
   418→      totalAtrasado,
   419→      previstoMes,
   420→      totalCobrancas: cobrancas.length,
   421→    };
   422→  }, [cobrancas]);
   423→
   424→  // Filtragem
   425→  const cobrancasFiltradas = useMemo(() => {
   426→    let lista = cobrancas;
   427→    if (filtroStatus !== "TODOS")
   428→      lista = lista.filter((c) => c.status === filtroStatus);
   429→    if (filtroTipo !== "TODOS") lista = lista.filter((c) => c.tipo === filtroTipo);
   430→    if (filtroSistema !== "TODOS")
   431→      lista = lista.filter((c) => c.sistemaId === filtroSistema);
   432→    if (busca.trim()) {
   433→      const termo = busca.toLowerCase();
   434→      lista = lista.filter(
   435→        (c) =>
   436→          c.sistemaNome.toLowerCase().includes(termo) ||
   437→          c.descricao.toLowerCase().includes(termo) ||
   438→          c.observacoes.toLowerCase().includes(termo)
   439→      );
   440→    }
   441→    return lista;
   442→  }, [cobrancas, filtroStatus, filtroTipo, filtroSistema, busca]);
   443→
   444→  const getStatusInfo = (status: StatusCobranca) =>
   445→    STATUS_COBRANCA.find((s) => s.valor === status) || STATUS_COBRANCA[0];
   446→  const getTipoInfo = (tipo: TipoCobranca) =>
   447→    TIPOS_COBRANCA.find((t) => t.valor === tipo) || TIPOS_COBRANCA[0];
   448→  const getFormaInfo = (forma: FormaPagamentoAdmin) =>
   449→    FORMAS_PAGAMENTO_ADMIN.find((f) => f.valor === forma);
   450→
   451→  const handleSalvarNova = useCallback(
   452→    (dados: Omit<Cobranca, "id" | "criadoEm">) => {
   453→      adicionarCobranca(dados);
   454→      setDialogNova(false);
   455→      toast.success("Cobranca gerada com sucesso!");
   456→    },
   457→    [adicionarCobranca]
   458→  );
   459→
   460→  const handleSalvarEdicao = useCallback(
   461→    (dados: Omit<Cobranca, "id" | "criadoEm">) => {
   462→      if (!dialogEditCobranca) return;
   463→      editarCobranca(dialogEditCobranca.id, dados);
   464→      setDialogEditCobranca(null);
   465→      toast.success("Cobranca atualizada!");
   466→    },
   467→    [dialogEditCobranca, editarCobranca]
   468→  );
   469→
   470→  const handleConfirmarPagamento = useCallback(
   471→    (id: string, dataPag: string, forma: FormaPagamentoAdmin) => {
   472→      registrarPagamento(id, dataPag, forma);
   473→      setDialogPagamento(null);
   474→      toast.success("Pagamento registrado!");
   475→    },
   476→    [registrarPagamento]
   477→  );
   478→
   479→  const handleRemover = useCallback(
   480→    (id: string) => {
   481→      removerCobranca(id);
   482→      setConfirmaRemoverCob(null);
   483→      toast.success("Cobranca removida.");
   484→    },
   485→    [removerCobranca]
   486→  );
   487→
   488→  const enviarLembreteWhatsApp = (cobranca: Cobranca) => {
   489→    const sistema = sistemas.find((s) => s.id === cobranca.sistemaId);
   490→    if (!sistema?.telefone) {
   491→      toast.error("Telefone nao cadastrado para este sistema.");
   492→      return;
   493→    }
   494→    const telLimpo = sistema.telefone.replace(/\D/g, "");
   495→    const numero = telLimpo.startsWith("55") ? telLimpo : `55${telLimpo}`;
   496→    const msg = encodeURIComponent(
   497→      `Ola! Aqui e o suporte do ZapFacil Pro.\n\nGostariamos de lhe lembrar sobre a cobranca:\n\n*${cobranca.descricao}*\nValor: ${formatarMoeda(cobranca.valor)}\nVencimento: ${formatarData(cobranca.dataVencimento)}\n\nPor favor, entre em contato para confirmar o pagamento. Estamos a disposicao!`
   498→    );
   499→    window.open(`https://wa.me/${numero}?text=${msg}`, "_blank");
   500→  };
   501→
   502→  // Cobranças do sistema selecionado para histórico
   503→  const cobrancasHistorico = dialogHistorico
   504→    ? cobrancas.filter((c) => c.sistemaId === dialogHistorico.id)
   505→    : [];
   506→
   507→  const totalPagoHistorico = cobrancasHistorico
   508→    .filter((c) => c.status === "PAGO")
   509→    .reduce((s, c) => s + c.valor, 0);
   510→  const totalPendenteHistorico = cobrancasHistorico
   511→    .filter((c) => c.status !== "PAGO" && c.status !== "CANCELADO")
   512→    .reduce((s, c) => s + c.valor, 0);
   513→
   514→  return (
   515→    <div className="space-y-5">
   516→      {/* Cards de estatísticas */}
   517→      <div className="grid grid-cols-2 lg:grid-cols-5 gap-3">
   518→        <Card className="border-0 shadow-sm">
   519→          <CardContent className="p-3.5">
   520→            <div className="flex items-center gap-1.5 mb-1">
   521→              <DollarSign className="w-3.5 h-3.5 text-emerald-500" />
   522→              <span className="text-[10px] text-gray-400 font-medium uppercase tracking-wider">
   523→                Recebido/mes
