    57→function SecaoConexao() {
    58→  const {
    59→    apiUrl, instanceName, apiKey, conectado,
    60→    configurarConexao, setConectado,
    61→  } = useZapBotStore();
    62→  const [editUrl, setEditUrl] = useState(apiUrl);
    63→  const [editInstance, setEditInstance] = useState(instanceName);
    64→  const [editKey, setEditKey] = useState(apiKey);
    65→  const [mostrarKey, setMostrarKey] = useState(false);
    66→  const [salvo, setSalvo] = useState(!!apiUrl);
    67→
    68→  const handleSalvar = () => {
    69→    if (!editUrl.trim() || !editInstance.trim()) {
    70→      toast.error("Preencha a URL e o nome da instancia.");
    71→      return;
    72→    }
    73→    configurarConexao(
    74→      editUrl.trim().replace(/\/$/, ""),
    75→      editInstance.trim(),
    76→      editKey.trim()
    77→    );
    78→    setSalvo(true);
    79→    toast.success("Configuracao salva!");
    80→  };
    81→
    82→  const handleConectar = async () => {
    83→    if (!apiUrl || !instanceName) {
    84→      toast.error("Configure a conexao primeiro.");
    85→      return;
    86→    }
    87→    toast.loading("Conectando ao WhatsApp...");
    88→    try {
    89→      const resp = await fetch(
    90→        `${apiUrl}/instance/connect/${instanceName}?apikey=${apiKey}`,
    91→        { method: "POST" }
    92→      );
    93→      const data = await resp.json();
    94→      if (data.code === 200 || data.connected) {
    95→        setConectado(true);
    96→        toast.dismiss();
    97→        toast.success("WhatsApp conectado com sucesso!");
    98→      } else if (data.base64) {
    99→        toast.dismiss();
   100→        toast.info("Escaneie o QR Code no WhatsApp para conectar.");
   101→        setConectado(true);
   102→      } else {
   103→        setConectado(false);
   104→        toast.dismiss();
   105→        toast.error("Erro ao conectar. Verifique os dados.");
   106→      }
   107→    } catch {
   108→      // Demo mode - conexao simulada
   109→      toast.dismiss();
   110→      setConectado(true);
   111→      toast.success("Modo demonstracao ativo! (Conexao simulada)");
   112→    }
   113→  };
   114→
   115→  const handleDesconectar = async () => {
   116→    try {
   117→      await fetch(
   118→        `${apiUrl}/instance/logout/${instanceName}?apikey=${apiKey}`,
   119→        { method: "DELETE" }
   120→      );
   121→    } catch { /* demo */ }
   122→    setConectado(false);
   123→    toast.info("WhatsApp desconectado.");
   124→  };
   125→
   126→  return (
   127→    <Card className="border-2 border-dashed border-emerald-200 dark:border-emerald-800">
   128→      <CardHeader className="pb-3">
   129→        <CardTitle className="flex items-center gap-2 text-sm">
   130→          <Server className="h-4 w-4 text-emerald-600" />
   131→          Conexao Evolution API
   132→          {conectado ? (
   133→            <Badge className="ml-auto bg-emerald-100 text-emerald-800 dark:bg-emerald-900 dark:text-emerald-200 text-[10px]">
   134→              <Wifi className="h-3 w-3 mr-1" /> Online
   135→            </Badge>
   136→          ) : (
   137→            <Badge variant="secondary" className="ml-auto text-[10px]">
   138→              <WifiOff className="h-3 w-3 mr-1" /> Offline
   139→            </Badge>
   140→          )}
   141→        </CardTitle>
   142→      </CardHeader>
   143→      <CardContent className="space-y-3">
   144→        {!salvo ? (
   145→          <>
   146→            <div className="space-y-2">
   147→              <Label className="text-xs font-semibold flex items-center gap-1.5">
   148→                <Globe className="h-3 w-3" /> URL da API
   149→              </Label>
   150→              <Input
   151→                placeholder="https://seu-servidor.com:8080"
   152→                value={editUrl}
   153→                onChange={(e) => setEditUrl(e.target.value)}
   154→                className="text-xs h-9"
   155→              />
   156→              <p className="text-[10px] text-muted-foreground">
   157→                URL do servidor onde a Evolution API esta rodando
   158→              </p>
   159→            </div>
   160→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   161→              <div className="space-y-1.5">
   162→                <Label className="text-xs font-semibold flex items-center gap-1.5">
   163→                  <Bot className="h-3 w-3" /> Nome da Instancia
   164→                </Label>
   165→                <Input
   166→                  placeholder="minha-empresa"
   167→                  value={editInstance}
   168→                  onChange={(e) => setEditInstance(e.target.value)}
   169→                  className="text-xs h-9"
   170→                />
   171→              </div>
   172→              <div className="space-y-1.5">
   173→                <Label className="text-xs font-semibold flex items-center gap-1.5">
   174→                  <Key className="h-3 w-3" /> API Key
   175→                </Label>
   176→                <div className="relative">
   177→                  <Input
   178→                    type={mostrarKey ? "text" : "password"}
   179→                    placeholder="Sua API Key"
   180→                    value={editKey}
   181→                    onChange={(e) => setEditKey(e.target.value)}
   182→                    className="text-xs h-9 pr-9"
   183→                  />
   184→                  <button
   185→                    type="button"
   186→                    onClick={() => setMostrarKey(!mostrarKey)}
   187→                    className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   188→                  >
   189→                    {mostrarKey ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   190→                  </button>
   191→                </div>
   192→              </div>
   193→            </div>
   194→            <Button
   195→              className="w-full h-9 bg-emerald-600 hover:bg-emerald-700 text-xs font-bold"
   196→              onClick={handleSalvar}
   197→            >
   198→              <Settings className="h-3.5 w-3.5 mr-1.5" />
   199→              Salvar Configuracao
   200→            </Button>
   201→          </>
   202→        ) : (
   203→          <div className="space-y-3">
   204→            <div className="bg-muted/50 rounded-lg p-3 space-y-1.5">
   205→              <div className="flex justify-between items-center">
   206→                <span className="text-[10px] text-muted-foreground uppercase font-bold">URL</span>
   207→                <Button
   208→                  variant="ghost"
   209→                  size="sm"
   210→                  className="h-6 text-[10px]"
   211→                  onClick={() => { navigator.clipboard.writeText(apiUrl); toast.success("URL copiada!"); }}
   212→                >
   213→                  <Copy className="h-3 w-3 mr-1" /> Copiar
   214→                </Button>
   215→              </div>
   216→              <p className="text-xs font-mono break-all">{apiUrl}</p>
   217→              <div className="flex gap-4">
   218→                <div>
   219→                  <span className="text-[10px] text-muted-foreground uppercase font-bold">Instancia</span>
   220→                  <p className="text-xs font-mono">{instanceName}</p>
   221→                </div>
   222→              </div>
   223→            </div>
   224→            <div className="grid grid-cols-2 gap-2">
   225→              <Button
   226→                className={`h-10 text-xs font-bold ${conectado ? "bg-red-500 hover:bg-red-600" : "bg-emerald-600 hover:bg-emerald-700"}`}
   227→                onClick={conectado ? handleDesconectar : handleConectar}
   228→              >
   229→                {conectado ? (
   230→                  <><PowerOff className="h-3.5 w-3.5 mr-1.5" /> Desconectar</>
   231→                ) : (
   232→                  <><Power className="h-3.5 w-3.5 mr-1.5" /> Conectar</>
   233→                )}
   234→              </Button>
   235→              <Button
   236→                variant="outline"
   237→                className="h-10 text-xs"
   238→                onClick={() => setSalvo(false)}
   239→              >
   240→                <Settings className="h-3.5 w-3.5 mr-1.5" />
   241→                Editar
   242→              </Button>
   243→            </div>
   244→          </div>
   245→        )}
   246→      </CardContent>
   247→    </Card>
   248→  );
   249→}

   358→function SecaoRespostas() {
   359→  const { respostas, adicionarResposta, editarResposta, toggleResposta, removerResposta } = useZapBotStore();
   360→  const [novoGatilho, setNovoGatilho] = useState("");
   361→  const [novaResposta, setNovaResposta] = useState("");
   362→  const [editId, setEditId] = useState<string | null>(null);
   363→  const [editGatilho, setEditGatilho] = useState("");
   364→  const [editRespostaVal, setEditRespostaVal] = useState("");
   365→
   366→  const handleAdicionar = () => {
   367→    if (!novoGatilho.trim() || !novaResposta.trim()) {
   368→      toast.error("Preencha o gatilho e a resposta.");
   369→      return;
   370→    }
   371→    adicionarResposta(novoGatilho, novaResposta);
   372→    setNovoGatilho("");
   373→    setNovaResposta("");
   374→    toast.success("Resposta adicionada!");
   375→  };
   376→
   377→  const handleEditar = (id: string) => {
   378→    if (!editGatilho.trim() || !editRespostaVal.trim()) return;
   379→    editarResposta(id, editGatilho, editRespostaVal);
   380→    setEditId(null);
   381→    toast.success("Resposta atualizada!");
   382→  };
   383→
   384→  const ativas = useMemo(() => respostas.filter((r) => r.ativo), [respostas]);
   385→
   386→  return (
   387→    <Card>
   388→      <CardHeader className="pb-3">
   389→        <CardTitle className="flex items-center gap-2 text-sm">
   390→          <Zap className="h-4 w-4 text-amber-500" />
   391→          Respostas Automaticas
   392→          <Badge variant="secondary" className="ml-auto text-[10px]">
   393→            {ativas.length} ativa{ativas.length !== 1 ? "s" : ""}
   394→          </Badge>
   395→        </CardTitle>
   396→      </CardHeader>
   397→      <CardContent className="space-y-3">
   398→        {/* Form novo */}
   399→        <div className="space-y-2">
   400→          <div className="grid grid-cols-1 sm:grid-cols-3 gap-2">
   401→            <Input
   402→              placeholder="Gatilho (ex: horario)"
   403→              value={novoGatilho}
   404→              onChange={(e) => setNovoGatilho(e.target.value)}
   405→              className="text-xs h-9 sm:col-span-1"
   406→              onKeyDown={(e) => e.key === "Enter" && handleAdicionar()}
   407→            />
   408→            <Input
   409→              placeholder="Resposta automatica..."
   410→              value={novaResposta}
   411→              onChange={(e) => setNovaResposta(e.target.value)}
   412→              className="text-xs h-9 sm:col-span-1"
   413→              onKeyDown={(e) => e.key === "Enter" && handleAdicionar()}
   414→            />
   415→            <Button
   416→              size="sm"
   417→              className="h-9 bg-amber-500 hover:bg-amber-600 text-xs"
   418→              onClick={handleAdicionar}
   419→            >
   420→              <Plus className="h-3 w-3 mr-1" /> Adicionar
   421→            </Button>
   422→          </div>
   423→          <p className="text-[10px] text-muted-foreground">
   424→            Quando o cliente digitar o gatilho, o bot responde automaticamente.
   425→          </p>
   426→        </div>
   427→
   428→        {/* Lista */}
   429→        <div className="space-y-1.5 max-h-72 overflow-y-auto">
   430→          {respostas.length === 0 && (
   431→            <p className="text-xs text-muted-foreground text-center py-4 italic">
   432→              Nenhuma resposta configurada.
   433→            </p>
   434→          )}
   435→          {respostas.map((r) => (
   436→            <div
   437→              key={r.id}
   438→              className={`flex items-start gap-2 p-2.5 rounded-lg border transition-colors ${
   439→                r.ativo
   440→                  ? "bg-white dark:bg-gray-900 border-gray-200 dark:border-gray-700"
   441→                  : "bg-gray-50 dark:bg-gray-900/50 border-gray-100 dark:border-gray-800 opacity-60"
   442→              }`}
   443→            >
   444→              {editId === r.id ? (
   445→                <div className="flex-1 space-y-1.5">
   446→                  <Input
   447→                    value={editGatilho}
   448→                    onChange={(e) => setEditGatilho(e.target.value)}
   449→                    className="text-xs h-7"
   450→                    placeholder="Gatilho"
   451→                  />
   452→                  <Input
   453→                    value={editRespostaVal}
   454→                    onChange={(e) => setEditRespostaVal(e.target.value)}
   455→                    className="text-xs h-7"
   456→                    placeholder="Resposta"
   457→                  />
   458→                  <div className="flex gap-1">
   459→                    <Button
   460→                      variant="ghost"
   461→                      size="sm"
   462→                      className="h-6 text-[10px] text-emerald-600"
   463→                      onClick={() => handleEditar(r.id)}
   464→                    >
   465→                      <CheckCircle2 className="h-3 w-3 mr-0.5" /> Salvar
   466→                    </Button>
   467→                    <Button
   468→                      variant="ghost"
   469→                      size="sm"
   470→                      className="h-6 text-[10px]"
   471→                      onClick={() => setEditId(null)}
   472→                    >
   473→                      <XCircle className="h-3 w-3 mr-0.5" /> Cancelar
   474→                    </Button>
   475→                  </div>
   476→                </div>
   477→              ) : (
   478→                <>
   479→                  <Switch
   480→                    checked={r.ativo}
   481→                    onCheckedChange={() => toggleResposta(r.id)}
   482→                    className="mt-1 shrink-0"
   483→                  />
   484→                  <div className="flex-1 min-w-0">
   485→                    <div className="flex items-center gap-2">
   486→                      <Badge variant="outline" className="text-[9px] font-mono shrink-0">
   487→                        {r.gatilho}
   488→                      </Badge>
   489→                      <p className="text-[11px] text-gray-600 dark:text-gray-400 truncate">
   490→                        {r.resposta}
   491→                      </p>
   492→                    </div>
   493→                  </div>
   494→                  <Button
   495→                    variant="ghost"
   496→                    size="sm"
   497→                    className="h-7 w-7 p-0 shrink-0"
   498→                    onClick={() => { setEditId(r.id); setEditGatilho(r.gatilho); setEditRespostaVal(r.resposta); }}
   499→                  >
   500→                    <Pencil className="h-3 w-3" />
   501→                  </Button>
   502→                  <Button
   503→                    variant="ghost"
   504→                    size="sm"
   505→                    className="h-7 w-7 p-0 text-red-500 shrink-0"
   506→                    onClick={() => { removerResposta(r.id); toast.success("Removida."); }}
   507→                  >
   508→                    <Trash2 className="h-3 w-3" />
   509→                  </Button>
   510→                </>
   511→              )}
   512→            </div>
   513→          ))}
   514→        </div>
   515→      </CardContent>
   516→    </Card>
   517→  );
   518→}

   592→                  {previewMenu.replace(/\*/g, "")}

   699→function SecaoDisparo() {
   700→  const { conectado, apiUrl, instanceName, apiKey } = useZapBotStore();
   701→  const { clientes } = useERPStore();
   702→  const [msgDisparo, setMsgDisparo] = useState("");
   703→  const [enviando, setEnviando] = useState(false);
   704→
   705→  const handleDisparar = async () => {
   706→    if (!msgDisparo.trim()) {
   707→      toast.error("Escreva a mensagem para disparar.");
   708→      return;
   709→    }
   710→    const clientesComTel = clientes.filter((c) => c.telefone);
   711→    if (clientesComTel.length === 0) {
   712→      toast.error("Nenhum cliente com telefone cadastrado.");
   713→      return;
   714→    }
   715→
   716→    setEnviando(true);
   717→    let sucesso = 0;
   718→    let falha = 0;
   719→
   720→    for (const cliente of clientesComTel) {
   721→      try {
   722→        const telLimpo = cliente.telefone.replace(/\D/g, "");
   723→        const numero = telLimpo.startsWith("55") ? telLimpo : `55${telLimpo}`;
   724→        await fetch(
   725→          `${apiUrl}/message/sendText/${instanceName}?apikey=${apiKey}`,
   726→          {
   727→            method: "POST",
   728→            headers: { "Content-Type": "application/json" },
   729→            body: JSON.stringify({
   730→              number: `${numero}@s.whatsapp.net`,
   731→              text: msgDisparo,
   732→            }),
   733→          }
   734→        );
   735→        sucesso++;
   736→      } catch {
   737→        falha++;
   738→      }
   739→    }
   740→
   741→    setEnviando(false);
   742→    if (falha === 0) {
   743→      toast.success(`${sucesso} mensagem(ns) enviada(s)!`);
   744→    } else {
   745→      toast.warning(`${sucesso} enviada(s), ${falha} falha(s).`);
   746→    }
   747→  };
   748→
   749→  return (
   750→    <Card>
   751→      <CardHeader className="pb-3">
   752→        <CardTitle className="flex items-center gap-2 text-sm">
   753→          <Send className="h-4 w-4 text-purple-500" />
   754→          Disparo em Massa
   755→          <Badge variant="secondary" className="ml-auto text-[10px]">
   756→            {clientes.filter((c) => c.telefone).length} contato(s)
   757→          </Badge>
   758→        </CardTitle>
   759→      </CardHeader>
   760→      <CardContent className="space-y-3">
   761→        <Textarea
   762→          value={msgDisparo}
   763→          onChange={(e) => setMsgDisparo(e.target.value)}
   764→          rows={4}
   765→          className="text-xs resize-none"
   766→          placeholder="Digite a mensagem que sera enviada para todos os clientes com telefone cadastrado..."
   767→        />
   768→        <div className="flex items-center justify-between">
   769→          <p className="text-[10px] text-muted-foreground">
   770→            Use *texto* para negrito e _texto_ para italico no WhatsApp.
   771→          </p>
   772→          <Button
   773→            size="sm"
   774→            className="h-9 bg-purple-600 hover:bg-purple-700 text-xs"
   775→            onClick={handleDisparar}
   776→            disabled={enviando || !msgDisparo.trim()}
   777→          >
   778→            {enviando ? (
   779→              <><RefreshCw className="h-3 w-3 mr-1 animate-spin" /> Enviando...</>
   780→            ) : (
   781→              <><Send className="h-3 w-3 mr-1" /> Disparar para Todos</>
   782→            )}
   783→          </Button>
   784→        </div>
   785→        {!conectado && (
   786→          <div className="bg-amber-50 dark:bg-amber-950/20 border border-amber-200 dark:border-amber-800 rounded-lg p-2.5 flex items-start gap-2">
   787→            <AlertCircle className="h-4 w-4 text-amber-600 mt-0.5 shrink-0" />
   788→            <p className="text-[10px] text-amber-700 dark:text-amber-400">
   789→              WhatsApp nao conectado. Configure a Evolution API na secao de conexao para habilitar o disparo real.
   790→            </p>
   791→          </div>
   792→        )}
   793→      </CardContent>
   794→    </Card>
   795→  );
   796→}
