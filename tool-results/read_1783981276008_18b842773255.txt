   355→        function checarLicencaValida() {
   356→            const chaveAtiva = localStorage.getItem('zap_licenca_chave') || "";
   357→            const telaBloqueio = document.getElementById('bloqueioAssinatura');
   358→            const labelValidade = document.getElementById('labelValidadeLicenca');
   359→
   360→            if (!chaveAtiva) {
   361→                telaBloqueio.style.display = "flex";
   362→                return false;
   363→            }
   364→
   365→            try {
   366→                // Decodificação simples base64 da chave para segurança básica contra leigos
   367→                const decodificada = atob(chaveAtiva); // Formato esperado: ZAPFACIL-AAAAMMDD
   368→                if (!decodificada.startsWith("ZAPFACIL-")) {
   369→                    telaBloqueio.style.display = "flex";
   370→                    return false;
   371→                }
   372→
   373→                const dataStr = decodificada.split("-")[1]; // Pega AAAAMMDD
   374→                const ano = parseInt(dataStr.substring(0,4));
   375→                const mes = parseInt(dataStr.substring(4,6)) - 1;
   376→                const dia = parseInt(dataStr.substring(6,8));
   377→
   378→                const dataValidade = new Date(ano, mes, dia, 23, 59, 59);
   379→                const hoje = new Date();
   380→
   381→                if (hoje > dataValidade) {
   382→                    telaBloqueio.style.display = "flex";
   383→                    return false;
   384→                } else {
   385→                    telaBloqueio.style.display = "none";
   386→                    labelValidade.innerText = `Licença ativa até: ${dia.toString().padStart(2,'0')}/${(mes+1).toString().padStart(2,'0')}/${ano}`;
   387→                    return true;
   388→                }
   389→            } catch (e) {
   390→                telaBloqueio.style.display = "flex";
   391→                return false;
   392→            }
   393→        }
