    57→function formatarMoeda(valor: number) {
    58→  return valor.toLocaleString("pt-BR", { style: "currency", currency: "BRL" });

    61→function formatarData(data: string) {
    62→  if (!data) return "";
    63→  const partes = data.split("-");
    64→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    65→  return data;
    66→}

    68→function diasRestantes(vencimento: string): number {
    69→  const hoje = new Date();
    70→  hoje.setHours(0, 0, 0, 0);
    71→  const venc = new Date(vencimento + "T00:00:00");
    72→  const diff = venc.getTime() - hoje.getTime();
    73→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    74→}

   500→function PainelAdminConteudo() {
   501→  const { sistemas, adicionarSistema, editarSistema, removerSistema } =
   502→    useAdminStore();
   503→  const [busca, setBusca] = useState("");
   504→  const [filtroStatus, setFiltroStatus] = useState<string>("TODOS");
   505→  const [filtroPlano, setFiltroPlano] = useState<string>("TODOS");
   506→  const [dialogForm, setDialogForm] = useState<SistemaCliente | null>(null);
   507→  const [dialogNovo, setDialogNovo] = useState(false);
   508→  const [dialogDetalhe, setDialogDetalhe] = useState<SistemaCliente | null>(null);
   509→  const [confirmaRemover, setConfirmaRemover] = useState<string | null>(null);
   510→
   511→  // Estatísticas
   512→  const stats = useMemo(() => {
   513→    const ativos = sistemas.filter((s) => s.status === "ATIVO").length;
   514→    const trials = sistemas.filter((s) => s.status === "TRIAL").length;
   515→    const expirados = sistemas.filter((s) => s.status === "EXPIRADO").length;
   516→    const receitaMensal = sistemas
   517→      .filter((s) => s.status === "ATIVO" || s.status === "TRIAL")
   518→      .reduce((s, v) => s + v.valorMensal, 0);
   519→    const vencendo = sistemas.filter(
   520→      (s) => s.status === "ATIVO" && diasRestantes(s.dataVencimento) <= 7 && diasRestantes(s.dataVencimento) > 0
   521→    ).length;
   522→    return { ativos, trials, expirados, receitaMensal, vencendo, total: sistemas.length };
   523→  }, [sistemas]);
   524→
   525→  // Filtragem
   526→  const sistemasFiltrados = useMemo(() => {
   527→    let lista = sistemas;
   528→    if (filtroStatus !== "TODOS") lista = lista.filter((s) => s.status === filtroStatus);
   529→    if (filtroPlano !== "TODOS") lista = lista.filter((s) => s.plano === filtroPlano);
   530→    if (busca.trim()) {
   531→      const termo = busca.toLowerCase();
   532→      lista = lista.filter(
   533→        (s) =>
   534→          s.empresa.toLowerCase().includes(termo) ||
   535→          s.responsavel.toLowerCase().includes(termo) ||
   536→          s.cidade.toLowerCase().includes(termo) ||
   537→          s.telefone.includes(termo) ||
   538→          s.email.toLowerCase().includes(termo)
   539→      );
