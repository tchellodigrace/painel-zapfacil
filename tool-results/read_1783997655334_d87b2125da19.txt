    73→    73→function diasRestantes(vencimento: string): number {
    74→    74→  const hoje = new Date();
    75→    75→  hoje.setHours(0, 0, 0, 0);
    76→    76→  const venc = new Date(vencimento + "T00:00:00");
    77→    77→  const diff = venc.getTime() - hoje.getTime();
    78→    78→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    79→    79→}
