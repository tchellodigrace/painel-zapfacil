     1→"use client";
     2→
     3→import { useState, useMemo } from "react";
     4→import { useERPStore } from "@/hooks/use-erp-store";
     5→import { Button } from "@/components/ui/button";
     6→import { Input } from "@/components/ui/input";
     7→import { Badge } from "@/components/ui/badge";
     8→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     9→import {
    10→  Dialog,
    11→  DialogContent,
    12→  DialogHeader,
    13→  DialogTitle,
    14→} from "@/components/ui/dialog";
    15→import { Separator } from "@/components/ui/separator";
    16→import type { Venda, Agendamento } from "@/types";
    17→import { STATUS_AGENDAMENTO } from "@/types";
    18→import {
    19→  Search,
    20→  ShoppingBag,
    21→  CalendarCheck,
    22→  Wallet,
    23→  ChevronRight,
    24→  Clock,
    25→  User,
    26→  CreditCard,
    27→  CheckCircle2,
    28→  AlertCircle,
    29→  ArrowLeft,
    30→  Receipt,
    31→} from "lucide-react";
    32→
    33→export function PortalCliente() {
    34→  const { vendas, agendamentos, empresa } = useERPStore();
    35→  const [busca, setBusca] = useState("");
    36→  const [clienteSelecionado, setClienteSelecionado] = useState<{
    37→    nome: string;
    38→    telefone: string;
    39→  } | null>(null);
    40→  const [vendaDetalhe, setVendaDetalhe] = useState<Venda | null>(null);
    41→
    42→  // Busca clientes únicos por telefone ou nome parcial
    43→  const clientesEncontrados = useMemo(() => {
    44→    if (busca.trim().length < 3) return [];
    45→    const termo = busca.trim().toLowerCase();
    46→    const mapa = new Map<
    47→      string,
    48→      { nome: string; telefone: string; doc: string }
    49→    >();
    50→
    51→    // Dos clientes cadastrados
    52→    const clientes = useERPStore.getState().clientes;
    53→    for (const c of clientes) {
    54→      const matchNome = c.nome.toLowerCase().includes(termo);
    55→      const matchTel = c.telefone.includes(termo);
    56→      const matchDoc = c.documento.includes(termo);
    57→      if (matchNome || matchTel || matchDoc) {
    58→        mapa.set(c.telefone, {
    59→          nome: c.nome,
    60→          telefone: c.telefone,
    61→          doc: c.documento,
    62→        });
    63→      }
    64→    }
    65→
    66→    // Das vendas (para pegar clientes que compraram mas podem não estar cadastrados)
    67→    for (const v of vendas) {
    68→      const matchNome = v.cliente.toLowerCase().includes(termo);
    69→      const matchDoc = v.docCliente.includes(termo);
    70→      if (matchNome || matchDoc) {
    71→        if (!mapa.has(v.docCliente)) {
    72→          mapa.set(v.docCliente, {
    73→            nome: v.cliente,
    74→            telefone: v.docCliente,
    75→            doc: v.docCliente,
    76→          });
    77→        }
    78→      }
    79→    }
    80→
    81→    // Dos agendamentos
    82→    for (const a of agendamentos) {
    83→      const matchNome = a.clienteNome.toLowerCase().includes(termo);
    84→      const matchTel = a.clienteTelefone.includes(termo);
    85→      if (matchNome || matchTel) {
    86→        if (!mapa.has(a.clienteTelefone)) {
    87→          mapa.set(a.clienteTelefone, {
    88→            nome: a.clienteNome,
    89→            telefone: a.clienteTelefone,
    90→            doc: "",
    91→          });
    92→        }
    93→      }
    94→    }
    95→
    96→    return Array.from(mapa.values());
    97→  }, [busca, vendas, agendamentos]);
    98→
    99→  // Dados do cliente selecionado
   100→  const dadosCliente = useMemo(() => {
   101→    if (!clienteSelecionado) return null;
   102→    const tel = clienteSelecionado.telefone;
   103→    const nome = clienteSelecionado.nome.toLowerCase();
   104→
   105→    const minhasVendas = vendas.filter(
   106→      (v) =>
   107→        v.docCliente === tel ||
   108→        v.cliente.toLowerCase() === nome
   109→    );
   110→    const meusAgendamentos = agendamentos.filter(
   111→      (a) =>
   112→        a.clienteTelefone === tel ||
   113→        a.clienteNome.toLowerCase() === nome
   114→    );
   115→
   116→    const totalGasto = minhasVendas
   117→      .filter((v) => v.status === "PAGO")
   118→      .reduce((s, v) => s + v.total, 0);
   119→    const totalPendente = minhasVendas
   120→      .filter((v) => v.status === "PENDENTE")
   121→      .reduce((s, v) => s + v.total, 0);
   122→    const totalVendas = minhasVendas.length;
   123→    const proxAgendamento = meusAgendamentos
   124→      .filter(
   125→        (a) =>
   126→          a.status !== "CONCLUIDO" &&
   127→          a.status !== "CANCELADO" &&
   128→          new Date(`${a.data}T${a.hora}`) >= new Date()
   129→      )
   130→      .sort(
   131→        (a, b) =>
   132→          new Date(`${a.data}T${a.hora}`).getTime() -
   133→          new Date(`${b.data}T${b.hora}`).getTime()
   134→      )[0];
   135→
   136→    return {
   137→      vendas: minhasVendas.sort(
   138→        (a, b) => b.timestamp - a.timestamp
   139→      ),
   140→      agendamentos: meusAgendamentos.sort(
   141→        (a, b) => b.timestamp - a.timestamp
   142→      ),
   143→      totalGasto,
   144→      totalPendente,
   145→      totalVendas,
   146→      proxAgendamento,
   147→    };
   148→  }, [clienteSelecionado, vendas, agendamentos]);
   149→
   150→  function formatarMoeda(valor: number) {
   151→    return valor.toLocaleString("pt-BR", {
   152→      style: "currency",
   153→      currency: "BRL",
   154→    });
   155→  }
   156→
   157→  function formatarData(data: string) {
   158→    if (!data) return "";
   159→    const partes = data.split("-");
   160→    if (partes.length === 3) {
   161→      return `${partes[2]}/${partes[1]}/${partes[0]}`;
   162→    }
   163→    return data;
   164→  }
   165→
   166→  function getStatusInfo(status: string) {
   167→    if (status === "PAGO")
   168→      return {
   169→        label: "Pago",
   170→        cor: "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300",
   171→        icone: CheckCircle2,
   172→      };
   173→    return {
   174→      label: "Pendente",
   175→      cor: "bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300",
   176→      icone: AlertCircle,
   177→    };
   178→  }
   179→
   180→  function getAgendamentoStatus(status: string) {
   181→    return (
   182→      STATUS_AGENDAMENTO.find((s) => s.valor === status) || {
   183→        label: status,
   184→        cor: "bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300",
   185→      }
   186→    );
   187→  }
   188→
   189→  // === TELA DE BUSCA ===
   190→  if (!clienteSelecionado) {
   191→    return (
   192→      <div className="min-h-screen bg-gradient-to-br from-slate-50 via-white to-emerald-50/30 dark:from-gray-950 dark:via-gray-900 dark:to-gray-950 flex flex-col">
   193→        {/* Header Portal */}
   194→        <header className="bg-white dark:bg-gray-900 border-b border-gray-200 dark:border-gray-800 sticky top-0 z-50">
   195→          <div className="max-w-2xl mx-auto flex items-center justify-between px-4 py-3">
   196→            <div className="flex items-center gap-3">
   197→              <img
   198→                src="/logo-empresa.png"
   199→                alt="Logo"
   200→                className="h-9 w-auto object-contain"
   201→              />
   202→            </div>
   203→            <a
   204→              href="/"
   205→              className="text-xs text-emerald-600 dark:text-emerald-400 hover:underline font-medium"
   206→            >
   207→              Acesso Admin
   208→            </a>
   209→          </div>
   210→        </header>
   211→
   212→        {/* Conteúdo busca */}
   213→        <div className="flex-1 flex flex-col items-center justify-center px-4 py-12">
   214→          <div className="w-full max-w-md space-y-6">
   215→            {/* Hero */}
   216→            <div className="text-center space-y-2">
   217→              <div className="inline-flex items-center justify-center w-14 h-14 rounded-2xl bg-emerald-100 dark:bg-emerald-900/30 mb-3">
   218→                <Receipt className="w-7 h-7 text-emerald-600 dark:text-emerald-400" />
   219→              </div>
   220→              <h1 className="text-2xl font-bold text-gray-900 dark:text-white">
   221→                Meus Pedidos
   222→              </h1>
   223→              <p className="text-sm text-gray-500 dark:text-gray-400 leading-relaxed">
   224→                Consulte seu historico de compras, agendamentos e
   225→                situacao de pagamento.
   226→              </p>
   227→            </div>
   228→
   229→            {/* Campo busca */}
   230→            <div className="relative">
   231→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   232→              <Input
   233→                placeholder="Digite seu nome, telefone ou CPF..."
   234→                value={busca}
   235→                onChange={(e) => setBusca(e.target.value)}
   236→                className="pl-10 h-12 text-base rounded-xl border-gray-200 dark:border-gray-700 bg-white dark:bg-gray-900 focus-visible:ring-emerald-500"
   237→              />
   238→            </div>
   239→
   240→            {/* Resultados */}
   241→            {busca.trim().length >= 3 &&
   242→              clientesEncontrados.length === 0 && (
   243→                <div className="text-center py-8">
   244→                  <User className="w-10 h-10 text-gray-300 dark:text-gray-600 mx-auto mb-2" />
   245→                  <p className="text-sm text-gray-400 dark:text-gray-500">
   246→                    Nenhum cliente encontrado com &quot;{busca}&quot;
   247→                  </p>
   248→                </div>
   249→              )}
   250→
   251→            {clientesEncontrados.length > 0 && (
   252→              <div className="space-y-2">
   253→                <p className="text-xs text-gray-400 px-1">
   254→                  {clientesEncontrados.length}{" "}
   255→                  {clientesEncontrados.length === 1
   256→                    ? "cliente encontrado"
   257→                    : "clientes encontrados"}
   258→                </p>
   259→                {clientesEncontrados.map((c, i) => (
   260→                  <button
   261→                    key={`${c.telefone}-${i}`}
   262→                    onClick={() => setClienteSelecionado(c)}
   263→                    className="w-full flex items-center gap-3 p-3.5 bg-white dark:bg-gray-900 border border-gray-200 dark:border-gray-700 rounded-xl hover:border-emerald-300 dark:hover:border-emerald-700 hover:shadow-sm transition-all text-left group"
   264→                  >
   265→                    <div className="flex items-center justify-center w-10 h-10 rounded-full bg-emerald-100 dark:bg-emerald-900/30 shrink-0">
   266→                      <User className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   267→                    </div>
   268→                    <div className="flex-1 min-w-0">
   269→                      <p className="font-semibold text-sm text-gray-900 dark:text-white truncate">
   270→                        {c.nome}
   271→                      </p>
   272→                      <p className="text-xs text-gray-400 truncate">
   273→                        {c.telefone}
   274→                      </p>
   275→                    </div>
   276→                    <ChevronRight className="w-4 h-4 text-gray-300 group-hover:text-emerald-500 transition-colors shrink-0" />
   277→                  </button>
   278→                ))}
   279→              </div>
   280→            )}
   281→          </div>
   282→        </div>
   283→      </div>
   284→    );
   285→  }
   286→
   287→  // === PAINEL DO CLIENTE ===
   288→  const statusAgendProx = clienteSelecionado
   289→    ? dadosCliente?.proxAgendamento
   290→      ? getAgendamentoStatus(dadosCliente.proxAgendamento.status)
   291→      : null
   292→    : null;
   293→
   294→  return (
   295→    <div className="min-h-screen bg-gradient-to-br from-slate-50 via-white to-emerald-50/30 dark:from-gray-950 dark:via-gray-900 dark:to-gray-950 flex flex-col">
   296→      {/* Header */}
   297→      <header className="bg-white dark:bg-gray-900 border-b border-gray-200 dark:border-gray-800 sticky top-0 z-50">
   298→        <div className="max-w-2xl mx-auto flex items-center gap-3 px-4 py-3">
   299→          <Button
   300→            variant="ghost"
   301→            size="icon"
   302→            className="h-8 w-8 shrink-0"
   303→            onClick={() => {
   304→              setClienteSelecionado(null);
   305→              setBusca("");
   306→            }}
   307→          >
   308→            <ArrowLeft className="h-4 w-4" />
   309→          </Button>
   310→          <div className="flex items-center gap-3 flex-1 min-w-0">
   311→            <div className="w-9 h-9 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center shrink-0">
   312→              <User className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   313→            </div>
   314→            <div className="min-w-0">
   315→              <p className="font-bold text-sm text-gray-900 dark:text-white truncate">
   316→                {clienteSelecionado.nome}
   317→              </p>
   318→              <p className="text-xs text-gray-400 truncate">
   319→                {clienteSelecionado.telefone}
   320→              </p>
   321→            </div>
   322→          </div>
   323→        </div>
   324→      </header>
   325→
   326→      {dadosCliente && (
   327→        <main className="flex-1 max-w-2xl mx-auto w-full p-4 space-y-4">
   328→          {/* Cards resumo */}
   329→          <div className="grid grid-cols-2 gap-3">
   330→            <Card className="border-0 shadow-sm bg-white dark:bg-gray-900">
   331→              <CardContent className="p-4">
   332→                <div className="flex items-center gap-2 mb-1">
   333→                  <Wallet className="w-4 h-4 text-emerald-500" />
   334→                  <span className="text-[11px] text-gray-400 font-medium uppercase tracking-wider">
   335→                    Total Pago
   336→                  </span>
   337→                </div>
   338→                <p className="text-xl font-bold text-gray-900 dark:text-white">
   339→                  {formatarMoeda(dadosCliente.totalGasto)}
   340→                </p>
   341→              </CardContent>
   342→            </Card>
   343→
   344→            <Card className="border-0 shadow-sm bg-white dark:bg-gray-900">
   345→              <CardContent className="p-4">
   346→                <div className="flex items-center gap-2 mb-1">
   347→                  <ShoppingBag className="w-4 h-4 text-blue-500" />
   348→                  <span className="text-[11px] text-gray-400 font-medium uppercase tracking-wider">
   349→                    Pedidos
   350→                  </span>
   351→                </div>
   352→                <p className="text-xl font-bold text-gray-900 dark:text-white">
   353→                  {dadosCliente.totalVendas}
   354→                </p>
   355→              </CardContent>
   356→            </Card>
   357→
   358→            <Card className="border-0 shadow-sm bg-white dark:bg-gray-900">
   359→              <CardContent className="p-4">
   360→                <div className="flex items-center gap-2 mb-1">
   361→                  <AlertCircle className="w-4 h-4 text-amber-500" />
   362→                  <span className="text-[11px] text-gray-400 font-medium uppercase tracking-wider">
   363→                    Pendente
   364→                  </span>
   365→                </div>
   366→                <p className="text-xl font-bold text-gray-900 dark:text-white">
   367→                  {formatarMoeda(dadosCliente.totalPendente)}
   368→                </p>
   369→              </CardContent>
   370→            </Card>
   371→
   372→            <Card className="border-0 shadow-sm bg-white dark:bg-gray-900">
   373→              <CardContent className="p-4">
   374→                <div className="flex items-center gap-2 mb-1">
   375→                  <CalendarCheck className="w-4 h-4 text-purple-500" />
   376→                  <span className="text-[11px] text-gray-400 font-medium uppercase tracking-wider">
   377→                    Agendamentos
   378→                  </span>
   379→                </div>
   380→                <p className="text-xl font-bold text-gray-900 dark:text-white">
   381→                  {dadosCliente.agendamentos.length}
   382→                </p>
   383→              </CardContent>
   384→            </Card>
   385→          </div>
   386→
   387→          {/* Próximo agendamento */}
   388→          {dadosCliente.proxAgendamento && (
   389→            <Card className="border-emerald-200 dark:border-emerald-800 bg-emerald-50/50 dark:bg-emerald-950/20">
   390→              <CardContent className="p-4">
   391→                <div className="flex items-start gap-3">
   392→                  <div className="w-10 h-10 rounded-xl bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center shrink-0 mt-0.5">
   393→                    <CalendarCheck className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   394→                  </div>
   395→                  <div className="flex-1 min-w-0">
   396→                    <p className="text-xs font-semibold text-emerald-700 dark:text-emerald-300 uppercase tracking-wider mb-1">
   397→                      Proximo Agendamento
   398→                    </p>
   399→                    <p className="font-bold text-gray-900 dark:text-white text-sm">
   400→                      {dadosCliente.proxAgendamento.servicoNome}
   401→                    </p>
   402→                    <div className="flex items-center gap-3 mt-1.5 flex-wrap">
   403→                      <span className="inline-flex items-center gap-1 text-xs text-gray-500 dark:text-gray-400">
   404→                        <Clock className="w-3 h-3" />
   405→                        {formatarData(dadosCliente.proxAgendamento.data)} às{" "}
   406→                        {dadosCliente.proxAgendamento.hora}
   407→                      </span>
   408→                      <span className="inline-flex items-center gap-1 text-xs text-gray-500 dark:text-gray-400">
   409→                        <User className="w-3 h-3" />
   410→                        {dadosCliente.proxAgendamento.colaboradorNome}
   411→                      </span>
   412→                    </div>
   413→                    <Badge
   414→                      className={`mt-2 text-[10px] font-semibold ${getAgendamentoStatus(dadosCliente.proxAgendamento.status).cor}`}
   415→                    >
   416→                      {getAgendamentoStatus(dadosCliente.proxAgendamento.status).label}
   417→                    </Badge>
   418→                  </div>
   419→                </div>
   420→              </CardContent>
   421→            </Card>
   422→          )}
   423→
   424→          {/* Histórico de vendas */}
   425→          <div>
   426→            <h2 className="text-sm font-bold text-gray-900 dark:text-white mb-3 flex items-center gap-2">
   427→              <ShoppingBag className="w-4 h-4 text-gray-400" />
   428→              Historico de Compras
   429→            </h2>
   430→
   431→            {dadosCliente.vendas.length === 0 ? (
   432→              <Card className="border-0 shadow-sm bg-white dark:bg-gray-900">
   433→                <CardContent className="p-8 text-center">
   434→                  <ShoppingBag className="w-10 h-10 text-gray-200 dark:text-gray-700 mx-auto mb-2" />
   435→                  <p className="text-sm text-gray-400">
   436→                    Nenhuma compra encontrada
   437→                  </p>
   438→                </CardContent>
   439→              </Card>
   440→            ) : (
   441→              <div className="space-y-2">
   442→                {dadosCliente.vendas.map((venda) => {
   443→                  const st = getStatusInfo(venda.status);
   444→                  const StatusIcone = st.icone;
   445→                  return (
   446→                    <button
   447→                      key={venda.id}
   448→                      onClick={() => setVendaDetalhe(venda)}
   449→                      className="w-full bg-white dark:bg-gray-900 border border-gray-200 dark:border-gray-700 rounded-xl p-4 hover:border-emerald-300 dark:hover:border-emerald-700 hover:shadow-sm transition-all text-left group"
   450→                    >
   451→                      <div className="flex items-start justify-between gap-3">
   452→                        <div className="flex-1 min-w-0">
   453→                          <div className="flex items-center gap-2 mb-1">
   454→                            <span className="text-xs text-gray-400">
   455→                              {formatarData(venda.data)} às {venda.hora}
   456→                            </span>
   457→                          </div>
   458→                          <p className="text-sm font-semibold text-gray-900 dark:text-white truncate">
   459→                            {venda.itens
   460→                              .map((i) => i.servicoNome)
   461→                              .join(", ")}
   462→                          </p>
   463→                          <p className="text-xs text-gray-400 mt-0.5">
   464→                            {venda.itens.length}{" "}
   465→                            {venda.itens.length === 1
   466→                              ? "item"
   467→                              : "itens"}
   468→                            {venda.colaboradorNome &&
   469→                              ` \u00b7 ${venda.colaboradorNome}`}
   470→                          </p>
   471→                        </div>
   472→                        <div className="text-right shrink-0">
   473→                          <p className="text-sm font-bold text-gray-900 dark:text-white">
   474→                            {formatarMoeda(venda.total)}
   475→                          </p>
   476→                          <Badge
   477→                            className={`mt-1 text-[10px] font-semibold ${st.cor}`}
   478→                          >
   479→                            <StatusIcone className="w-3 h-3 mr-1" />
   480→                            {st.label}
   481→                          </Badge>
   482→                        </div>
   483→                      </div>
   484→                    </button>
   485→                  );
   486→                })}
   487→              </div>
   488→            )}
   489→          </div>
   490→
   491→          {/* Agendamentos */}
   492→          {dadosCliente.agendamentos.length > 0 && (
   493→            <div>
   494→              <h2 className="text-sm font-bold text-gray-900 dark:text-white mb-3 flex items-center gap-2">
   495→                <CalendarCheck className="w-4 h-4 text-gray-400" />
   496→                Agendamentos
   497→              </h2>
   498→              <div className="space-y-2">
   499→                {dadosCliente.agendamentos.map((ag) => {
   500→                  const st = getAgendamentoStatus(ag.status);
   501→                  return (
   502→                    <Card
   503→                      key={ag.id}
   504→                      className="border-0 shadow-sm bg-white dark:bg-gray-900"
   505→                    >
   506→                      <CardContent className="p-4">
   507→                        <div className="flex items-start justify-between gap-3">
   508→                          <div className="flex-1 min-w-0">
   509→                            <p className="font-semibold text-sm text-gray-900 dark:text-white truncate">
   510→                              {ag.servicoNome}
   511→                            </p>
