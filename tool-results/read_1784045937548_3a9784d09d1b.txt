    74→    74→function formatarData(data: string) {

    81→    81→function diasRestantes(vencimento: string): number {
    82→    82→  const hoje = new Date();
    83→    83→  hoje.setHours(0, 0, 0, 0);
    84→    84→  const venc = new Date(vencimento + "T00:00:00");
    85→    85→  const diff = venc.getTime() - hoje.getTime();
    86→    86→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    87→    87→}
