    71→function formatarData(data: string) {
    72→  if (!data) return "";
    73→  const partes = data.split("-");
    74→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    75→  return data;
    76→}

    78→function diasRestantes(vencimento: string): number {
    79→  const hoje = new Date();
    80→  hoje.setHours(0, 0, 0, 0);
    81→  const venc = new Date(vencimento + "T00:00:00");
    82→  const diff = venc.getTime() - hoje.getTime();
    83→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    84→}

    89→function DialogTrocarSenha({
    90→  open,
    91→  onOpenChange,
    92→}: {
    93→  open: boolean;
    94→  onOpenChange: (v: boolean) => void;
    95→}) {
    96→  const { alterarSenha } = useAdminStore();
    97→  const [senhaAtual, setSenhaAtual] = useState("");
    98→  const [novaSenha, setNovaSenha] = useState("");
    99→  const [confirmarNova, setConfirmarNova] = useState("");
   100→  const [mostrarAtual, setMostrarAtual] = useState(false);
   101→  const [mostrarNova, setMostrarNova] = useState(false);
   102→
   103→  const handleSalvar = () => {
   104→    if (!senhaAtual || !novaSenha) {
   105→      toast.error("Preencha todos os campos.");
   106→      return;
   107→    }
   108→    if (novaSenha.length < 4) {
   109→      toast.error("A nova senha deve ter pelo menos 4 caracteres.");
   110→      return;
   111→    }
   112→    if (novaSenha !== confirmarNova) {
   113→      toast.error("As senhas nao conferem.");
   114→      return;
   115→    }
   116→    const ok = alterarSenha(senhaAtual, novaSenha);
   117→    if (ok) {
   118→      toast.success("Senha alterada com sucesso!");
   119→      setSenhaAtual("");
   120→      setNovaSenha("");
   121→      setConfirmarNova("");
   122→      onOpenChange(false);
   123→    } else {
   124→      toast.error("Senha atual incorreta.");
   125→    }
   126→  };
   127→
   128→  return (
   129→    <Dialog open={open} onOpenChange={onOpenChange}>
   130→      <DialogContent className="max-w-sm">
   131→        <DialogHeader>
   132→          <DialogTitle className="flex items-center gap-2 text-base">
   133→            <KeyRound className="h-5 w-5 text-gray-600" />

   135→          </DialogTitle>
   136→        </DialogHeader>
   137→        <div className="space-y-4">
   138→          <div className="space-y-1.5">
   139→            <Label className="text-xs font-medium">Senha atual</Label>
   140→            <div className="relative">
   141→              <Input
   142→                type={mostrarAtual ? "text" : "password"}
   143→                value={senhaAtual}
   144→                onChange={(e) => setSenhaAtual(e.target.value)}
   145→                placeholder="Digite a senha atual"
   146→                className="pr-10 h-10 text-sm"
   147→                onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   148→              />
   149→              <button
   150→                type="button"
   151→                onClick={() => setMostrarAtual(!mostrarAtual)}
   152→                className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600"
   153→              >
   154→                {mostrarAtual ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   155→              </button>
   156→            </div>
   157→          </div>
   158→          <div className="space-y-1.5">
   159→            <Label className="text-xs font-medium">Nova senha</Label>
   160→            <Input
   161→              type="password"
   162→              value={novaSenha}
   163→              onChange={(e) => setNovaSenha(e.target.value)}
   164→              placeholder="Minimo 4 caracteres"
   165→              className="h-10 text-sm"
   166→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   167→            />
   168→          </div>
   169→          <div className="space-y-1.5">
   170→            <Label className="text-xs font-medium">Confirmar nova senha</Label>
   171→            <Input
   172→              type="password"
   173→              value={confirmarNova}
   174→              onChange={(e) => setConfirmarNova(e.target.value)}
   175→              placeholder="Repita a nova senha"
   176→              className="h-10 text-sm"
