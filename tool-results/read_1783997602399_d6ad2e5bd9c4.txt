    66→function formatarData(data: string) {
    67→  if (!data) return "";
    68→  const partes = data.split("-");
    69→  if (partes.length === 3) return `${partes[2]}/${partes[1]}/${partes[0]}`;
    70→  return data;
    71→}

    73→function diasRestantes(vencimento: string): number {
    74→  const hoje = new Date();
    75→  hoje.setHours(0, 0, 0, 0);
    76→  const venc = new Date(vencimento + "T00:00:00");
    77→  const diff = venc.getTime() - hoje.getTime();
    78→  return Math.ceil(diff / (1000 * 60 * 60 * 24));
    79→}

    84→function DialogTrocarSenha({
    85→  open,
    86→  onOpenChange,
    87→}: {
    88→  open: boolean;
    89→  onOpenChange: (v: boolean) => void;
    90→}) {
    91→  const { alterarSenha } = useAdminStore();
    92→  const [senhaAtual, setSenhaAtual] = useState("");
    93→  const [novaSenha, setNovaSenha] = useState("");
    94→  const [confirmarNova, setConfirmarNova] = useState("");
    95→  const [mostrarAtual, setMostrarAtual] = useState(false);
    96→  const [mostrarNova, setMostrarNova] = useState(false);
    97→
    98→  const handleSalvar = () => {
    99→    if (!senhaAtual || !novaSenha) {
   100→      toast.error("Preencha todos os campos.");
   101→      return;
   102→    }
   103→    if (novaSenha.length < 4) {
   104→      toast.error("A nova senha deve ter pelo menos 4 caracteres.");
   105→      return;
   106→    }
   107→    if (novaSenha !== confirmarNova) {
   108→      toast.error("As senhas nao conferem.");
   109→      return;
   110→    }
   111→    const ok = alterarSenha(senhaAtual, novaSenha);
   112→    if (ok) {
   113→      toast.success("Senha alterada com sucesso!");
   114→      setSenhaAtual("");
   115→      setNovaSenha("");
   116→      setConfirmarNova("");
   117→      onOpenChange(false);
   118→    } else {
   119→      toast.error("Senha atual incorreta.");
   120→    }
   121→  };
   122→
   123→  return (
   124→    <Dialog open={open} onOpenChange={onOpenChange}>
   125→      <DialogContent className="max-w-sm">
   126→        <DialogHeader>
   127→          <DialogTitle className="flex items-center gap-2 text-base">
   128→            <KeyRound className="h-5 w-5 text-gray-600" />
   129→            Alterar Senha
   130→          </DialogTitle>
   131→        </DialogHeader>
   132→        <div className="space-y-4">
   133→          <div className="space-y-1.5">
   134→            <Label className="text-xs font-medium">Senha atual</Label>
   135→            <div className="relative">
   136→              <Input
   137→                type={mostrarAtual ? "text" : "password"}
   138→                value={senhaAtual}
   139→                onChange={(e) => setSenhaAtual(e.target.value)}
   140→                placeholder="Digite a senha atual"
   141→                className="pr-10 h-10 text-sm"
   142→                onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   143→              />
   144→              <button
   145→                type="button"
   146→                onClick={() => setMostrarAtual(!mostrarAtual)}
   147→                className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600"
   148→              >
   149→                {mostrarAtual ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   150→              </button>
   151→            </div>
   152→          </div>
   153→          <div className="space-y-1.5">
   154→            <Label className="text-xs font-medium">Nova senha</Label>
   155→            <Input
   156→              type="password"
   157→              value={novaSenha}
   158→              onChange={(e) => setNovaSenha(e.target.value)}
   159→              placeholder="Minimo 4 caracteres"
   160→              className="h-10 text-sm"
   161→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   162→            />
   163→          </div>
   164→          <div className="space-y-1.5">
   165→            <Label className="text-xs font-medium">Confirmar nova senha</Label>
   166→            <Input
   167→              type="password"
   168→              value={confirmarNova}
   169→              onChange={(e) => setConfirmarNova(e.target.value)}
   170→              placeholder="Repita a nova senha"
   171→              className="h-10 text-sm"
   172→              onKeyDown={(e) => e.key === "Enter" && handleSalvar()}
   173→            />
   174→          </div>
   175→          <div className="flex gap-2 justify-end pt-1">
   176→            <Button variant="ghost" size="sm" className="text-xs" onClick={() => onOpenChange(false)}>
   177→              Cancelar
   178→            </Button>
   179→            <Button size="sm" className="text-xs bg-emerald-600 hover:bg-emerald-700" onClick={handleSalvar}>
   180→              Salvar
   181→            </Button>
