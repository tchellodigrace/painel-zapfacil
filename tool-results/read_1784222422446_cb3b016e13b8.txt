    27→const AUTH_KEY = "zapfacil_auth";
    28→const SESSION_KEY = "zapfacil_session";
    29→
    30→interface Credenciais {
    31→  email: string;
    32→  senha: string;
    33→  nomeEmpresa: string;
    34→  nomeResponsavel: string;
    35→  telefone: string;
    36→}
    37→
    38→function carregarCredenciais(): Credenciais | null {
    39→  if (typeof window === "undefined") return null;
    40→  try {
    41→    const item = localStorage.getItem(AUTH_KEY);
    42→    return item ? (JSON.parse(item) as Credenciais) : null;
    43→  } catch {
    44→    return null;
    45→  }
    46→}
    47→
    48→function salvarCredenciais(cred: Credenciais): void {
    49→  if (typeof window === "undefined") return;
    50→  localStorage.setItem(AUTH_KEY, JSON.stringify(cred));
    51→}
    52→
    53→function criarSessao(): void {
    54→  if (typeof window === "undefined") return;
    55→  sessionStorage.setItem(SESSION_KEY, "autenticado");
    56→}
    57→
    58→function verificarSessao(): boolean {
    59→  if (typeof window === "undefined") return false;
    60→  return sessionStorage.getItem(SESSION_KEY) === "autenticado";
    61→}
    62→
    63→function destruirSessao(): void {
    64→  if (typeof window === "undefined") return;
    65→  sessionStorage.removeItem(SESSION_KEY);
    66→}
